lw x1,0(x0) // x1 = 0x400 base MMIO
lw x2,4(x0) // x2 = 1 incremento
lw x3,8(x0) // x3 = 8 mascara KEY[3]
lw x4,12(x0) // x4 = 4 mascara KEY[2]
lw x5,16(x0) // x5 = 2 mascara KEY[1]
add x10,x0,x0 // contador = 0
lw x7,4(x1) // prev = KEY (amostra inicial, evita borda falsa)
sw x10,12(x1) // LOOP: LEDG = contador (hw usa so os 9 LSB)
lw x6,4(x1) // cur = KEY
and x12,x6,x3 // KEY3: cur & 8
and x13,x7,x3 // KEY3: prev & 8
beq x12,x0,8 // cur==0 (pressionado)? -> testa prev
beq x0,x0,16 // solto -> CHK2
beq x13,x3,8 // prev==1 (estava solto)? -> borda de descida
beq x0,x0,8 // sem borda -> CHK2
add x10,x10,x2 // contador = contador + 1
and x12,x6,x4 // CHK2: cur & 4
and x13,x7,x4 // CHK2: prev & 4
beq x12,x0,8 // cur==0 (pressionado)? -> testa prev
beq x0,x0,16 // solto -> CHK1
beq x13,x4,8 // prev==1 (estava solto)? -> borda de descida
beq x0,x0,8 // sem borda -> CHK1
sub x10,x10,x2 // contador = contador - 1
and x12,x6,x5 // CHK1: cur & 2
and x13,x7,x5 // CHK1: prev & 2
beq x12,x0,8 // cur==0 (pressionado)? -> testa prev
beq x0,x0,16 // solto -> NEXT
beq x13,x5,8 // prev==1 (estava solto)? -> borda de descida
beq x0,x0,8 // sem borda -> NEXT
add x10,x0,x0 // contador = 0
add x7,x6,x0 // NEXT: prev = cur
beq x0,x0,-96 // volta para LOOP
